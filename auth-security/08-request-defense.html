Input is data, never code

✓Always pass values as arguments (? / $1). The driver sends them separately — they can't change the query.
→GORM's Where("email = ?", e) is safe; Where("email = " + e) is not.
→Column names and ORDER BY can't be placeholders — pick them from an allowlist.

func decode(w http.ResponseWriter, r *http.Request,
    v any) error {
    r.Body = http.MaxBytesReader(w, r.Body, 1<<20) // 1 MB
    dec := json.NewDecoder(r.Body)
    dec.DisallowUnknownFields()
    return dec.Decode(v)
}

func (l LoginReq) Validate() error {
    var errs []error
    if _, err := mail.ParseAddress(l.Email); err != nil {
        errs = append(errs, errors.New("email: invalid"))
    }
    if n := len(l.Password); n < 8 || n > 72 {
        errs = append(errs, errors.New("password: 8-72 bytes"))
    }
    return errors.Join(errs...)
}

var tooBig *http.MaxBytesError

→Go's default is no size limit — without MaxBytesReader one request can fill your memory.
→Validate, don't sanitize, on the way in. Escape on the way out — HTML templates, SQL placeholders.
→Unknown-field rejection stops "role":"admin" sneaking into a struct.

8c

A token bucket on /login

Five tokens, one back every 12 seconds — 5 per minute per IP. Each frame is real x/time/rate output.

internal/httpx/ratelimit.go
type IPLimiter struct {
    mu  sync.Mutex
    ips map[string]*rate.Limiter
    r   rate.Limit
    b   int
}

func (l *IPLimiter) get(ip string) *rate.Limiter {
    l.mu.Lock()
    defer l.mu.Unlock()
    lim, ok := l.ips[ip]
    if !ok {
        lim = rate.NewLimiter(l.r, l.b)
        l.ips[ip] = lim
    }
    return lim
}

func (l *IPLimiter) Middleware(next http.Handler) http.Handler {
    fn := func(w http.ResponseWriter, r *http.Request) {
        ip, _, _ := net.SplitHostPort(r.RemoteAddr)
        if !l.get(ip).Allow() {
            w.Header().Set("Retry-After", "12")
            http.Error(w, "too many requests", 429)
            return
        }
        next.ServeHTTP(w, r)
    }
    return http.HandlerFunc(fn)
}

// 5 per minute = one token every 12 s, burst 5
lim := NewIPLimiter(time.Minute/5, 5)
mux.Handle("POST /login", lim.Middleware(login))

func Allow(ctx context.Context, rdb *redis.Client,
    ip string, limit int64, win time.Duration) (bool, error) {
    key := "rl:login:" + ip
    n, err := rdb.Incr(ctx, key).Result()
    if err != nil {
        return false, err
    }
    if n == 1 {
        rdb.Expire(ctx, key, win) // first hit opens the window
    }
    return n <= limit, nil
}

→In-memory limits are per process: 3 instances = 3× the limit. Use Redis once you scale out.
→Behind a load balancer RemoteAddr is the proxy — read X-Forwarded-For only from a proxy you trust.
→Also limit per account (email): a botnet with 1,000 IPs still hits one user's bucket.
→The ips map grows forever — evict idle entries on a timer in production.
