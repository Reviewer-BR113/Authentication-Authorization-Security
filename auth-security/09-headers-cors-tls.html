9a

The preflight: the browser asks first

Before a cross-origin PATCH with a token, the browser sends an OPTIONS and waits for permission.

internal/httpx/cors.go
func CORS(allowed []string, next http.Handler) http.Handler {
    fn := func(w http.ResponseWriter, r *http.Request) {
        origin := r.Header.Get("Origin")
        w.Header().Add("Vary", "Origin")
        if origin != "" && slices.Contains(allowed, origin) {
            h := w.Header()
            h.Set("Access-Control-Allow-Origin", origin)
            h.Set("Access-Control-Allow-Credentials", "true")
            if r.Method == http.MethodOptions {
                h.Set("Access-Control-Allow-Methods",
                    "GET, POST, PATCH, DELETE")
                h.Set("Access-Control-Allow-Headers",
                    "Authorization, Content-Type")
                h.Set("Access-Control-Max-Age", "600")
                w.WriteHeader(http.StatusNoContent)
                return
            }
        }
        if r.Method == http.MethodOptions {
            w.WriteHeader(http.StatusNoContent) // no CORS headers
            return
        }
        next.ServeHTTP(w, r)
    }
    return http.HandlerFunc(fn)
}

✗Allow-Origin: * with credentials is rejected by browsers — and reflecting any Origin back is worse. Use an exact allowlist.
→Vary: Origin stops a cache serving one origin's CORS answer to another.
→CORS is not access control. The browser enforces it; curl ignores it. Auth still runs on every request.

internal/httpx/headers.go
func SecureHeaders(next http.Handler) http.Handler {
    fn := func(w http.ResponseWriter, r *http.Request) {
        h := w.Header()
        h.Set("X-Content-Type-Options", "nosniff")
        h.Set("X-Frame-Options", "DENY")
        h.Set("Content-Security-Policy",
            "default-src 'none'; frame-ancestors 'none'")
        h.Set("Referrer-Policy", "no-referrer")
        h.Set("Strict-Transport-Security",
            "max-age=63072000; includeSubDomains")
        next.ServeHTTP(w, r)
    }
    return http.HandlerFunc(fn)
}
→A JSON API can use the strictest CSP — default-src 'none' — because it serves no HTML.
→Set headers before next.ServeHTTP — once the body starts, headers are sent.
